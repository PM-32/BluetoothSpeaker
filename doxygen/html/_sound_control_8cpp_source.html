79 // Функция для записи звука в I2S
80 size_t write(const uint8_t *audioData, size_t lengthData) override
81 {
82 // Указатель на данные (сэмплы)
83 int16_t *samples = (int16_t*) audioData;
84
85 // Количество сэмплов
86 size_t samplesQuantity = lengthData / AUDIO_CHANNELS_QUANTITY;
87
88 // Вычисление коэффициента громкости с учетом ограничителя
89 float volumeScaleCoeff = volumeCoeff * LIMITER_COEFF;
90
91 for (size_t sampleIndex = 0; sampleIndex < samplesQuantity; sampleIndex++)
92 {
93 // Определение канала (четные - левый, нечетные - правый)
94 uint8_t channel = (0 == sampleIndex % AUDIO_CHANNELS_QUANTITY) ? 0 : 1;
95
96 // Применение эквалайзера к сэмплу
97 int16_t equalizerSample = SoundPresets_ProcessSample(samples[sampleIndex], channel);
98
99 // Масштабирование громкости сэмпла
100 float scaledVolumeSample = (float) equalizerSample * volumeScaleCoeff;
101
102 // Ограничение громкости для 16-ти битного сэмпла
103 if (scaledVolumeSample > MAX_VOLUME_SAMPLE)
104 {
105 scaledVolumeSample = MAX_VOLUME_SAMPLE;
106 }
107 else if (scaledVolumeSample < MIN_VOLUME_SAMPLE)
108 {
109 scaledVolumeSample = MIN_VOLUME_SAMPLE;
110 }
111
112 // Сохранение нового сэмпла с отмасштабированной громкостью
113 samples[sampleIndex] = (int16_t) scaledVolumeSample;
114 }
115
116 return audioOut.write(audioData, lengthData);
117 }

121 void SetVolume(uint8_t percents)
122 {
123 // Ограничение громкости звука в процентах
124 if (percents > MAX_VOLUME_PERCENT)
125 {
126 percents = MAX_VOLUME_PERCENT;
127 }
128
129 // Примечание. Для регулировки громкости используется корневая
130 // функция. Она нужна для быстрого роста громкости в начале
131 // диапазона. Это компенсирует логарифмическое восприятие слуха
132
133 // Нормализованное значение громкости
134 float normalizedVolume = (float) percents / MAX_VOLUME_PERCENT;
135
136 // Расчет коэффициента громкости
137 volumeCoeff = powf(normalizedVolume, VOLUME_CURVE_SHAPE);
138 }

144

146 float volumeCoeff = 1.0f;

152static void VolumeProtectionCheck(uint8_t volumePercent)
153{
154 // Получение текущего пресета
155 EqualizerPreset currentPreset = SoundPresets_GetCurrentPreset();
156
157 // Если выбран опасный пресет (BASS или ELECTRO)
158 // и громкость превышает безопасный уровень
159 if ((EQUALIZER_PRESET_BASS == currentPreset || EQUALIZER_PRESET_ELECTRO == currentPreset) &&
160 (volumePercent > MAX_SAFE_VOLUME_PERCENT))
161 {
162 #ifdef DEBUG_INFO_VOLUME_CONTROL
163
164 Serial.printf("Защита: громкость %u%% превышает безопасный уровень %u%% для пресета %s. Переключение на NORMAL.\r\n",
165 volumePercent, MAX_SAFE_VOLUME_PERCENT,
166 (EQUALIZER_PRESET_BASS == currentPreset) ? "BASS" : "ELECTRO");
167
168 #endif // DEBUG_INFO_VOLUME_CONTROL
169
170 // Переключение на безопасный пресет
171 SoundPresets_SetPreset(EQUALIZER_PRESET_NORMAL);
172 }
173}

178static void VolumeChangeCallback(int volume)
179{
180 // Ограничение максимальной громкости на уровне 90%
181 if (volume > MAX_LIMITED_VOLUME_AVRCP)

191 volumeControlSource = VOLUME_CONTROL_SOURCE_PHONE;

195void SoundControl_Init(void)
196{
197 auto cfg = i2s.defaultConfig(); // Установка конфигурации по умолчанию
198
199 cfg.port_no = I2S_NUMBER; // Номер используемого I2S (I2S0 или I2S1)
200 cfg.pin_bck = I2S_BCK_PIN; // Номер пина BCK (тактовый сигнал)
201 cfg.pin_ws = I2S_WS_PIN; // Номер пина WS/LCK (выбор канала: правый/левый)
202 cfg.pin_data = I2S_DIN_PIN; // Номер пина DIN (аудиоданные)
203 cfg.sample_rate = SAMPLE_RATE; // Частота дискретизации (Гц)
204 cfg.bits_per_sample = 16; // Количество бит для цифрового представления одного сэмпла (отсчета) звука
205 cfg.channels = AUDIO_CHANNELS_QUANTITY; // Количество каналов (моно = 1/стерео = 2)
206
207 // Запуск I2S1
208 i2s.begin(cfg);
209
210 // Включение регулировки громкости
211 a2dp_sink.set_output(volumeStream);
212
213 // Регистрация коллбэк-функции для прерывания
214 // по изменению громкости со смартфона
215 a2dp_sink.set_avrc_rn_volumechange(VolumeChangeCallback);
216
217 // Запуск A2DP-протокола
218 a2dp_sink.start(SPEAKER_NAME);
219
220 // Получение громкости звука со смартфона при старте
221 uint8_t startVolume = (uint8_t) a2dp_sink.get_volume();
222
223 // Ограничение стартовой громкости
224 if (startVolume > MAX_LIMITED_VOLUME_AVRCP)
225 {
226 startVolume = MAX_LIMITED_VOLUME_AVRCP;
227 }
228
229 // Вычисление стартовой громкости в процентах
230 currentVolumeInPercents = (uint8_t) (startVolume / VOLUME_PERCENT_TO_AVRCP_COEFF);
231
232 // Сохранение значения громкости звука,
233 // установленной со смартфона, в процентах
234 lastPhoneVolumeInPercents = currentVolumeInPercents;
235
236 // Установка громкости звука со смартфона
237 volumeStream.SetVolume(currentVolumeInPercents);
238
239 // Инициализация модуля звуковых пресетов
240 SoundPresets_Init();
241}

310void SoundControl_Volume(void)
311{
312 // Если текущим источником установки громкости звука является смартфон
313 if (volumeControlSource == VOLUME_CONTROL_SOURCE_PHONE)
314 {
315 // Если получено новое значение громкости со
316 // смартфона и оно отличается от предыдущего
317 if ((lastPhoneVolumeInPercents <= MAX_LIMITED_VOLUME_PERCENT) &&
318 (lastPhoneVolumeInPercents != currentVolumeInPercents))
319 {
320 // Сохранение текущей громкости звука
321 currentVolumeInPercents = lastPhoneVolumeInPercents;
322
323 // Защита от опасной комбинации громкости и пресета
324 VolumeProtectionCheck(currentVolumeInPercents);
325
326 // Установка громкости звука
327 volumeStream.SetVolume(currentVolumeInPercents);
328
329 #ifdef DEBUG_INFO_VOLUME_CONTROL
330
331 Serial.printf("Громкость, установленная смартфоном: %u%%\r\n", currentVolumeInPercents);
332
333 #endif // DEBUG_INFO_VOLUME_CONTROL
334 }
335 }
336
337 // Если текущим источником установки громкости звука является потенциометр
338 if (volumeControlSource == VOLUME_CONTROL_SOURCE_POTENTIOMETER)
339 {
340 // Если потенциометр не вращался более VOLUME_TIMEOUT_PERIOD
341 if ((UserTimer_GetCounterTime() - lastPotentiometerChangePositionTime) > VOLUME_TIMEOUT_PERIOD)
342 {
343 // Возврат управления громкостью звука к смартфону
344 volumeControlSource = VOLUME_CONTROL_SOURCE_PHONE;
345
346 // Расчет громкости звука в рамках протокола AVRCP
347 uint8_t avrcpVolume = (uint8_t) (currentVolumeInPercents * VOLUME_PERCENT_TO_AVRCP_COEFF);
348
349 // Обновление текущей громкости звука
350 // смартфоне для синхронизации значения
351 a2dp_sink.set_volume(avrcpVolume);
352
353 // Обновление последнего значения со смартфона
354 lastPhoneVolumeInPercents = currentVolumeInPercents;
355
356 #ifdef DEBUG_INFO_VOLUME_CONTROL
357
358 Serial.printf("Управление громкостью вернулось к смартфону, текущая громкость: %u%%\r\n", currentVolumeInPercents);
359
360 #endif // DEBUG_INFO_VOLUME_CONTROL
361 }
362 }
363
364 // Получение отсчетов АЦП потенциометра в процентах
365 uint8_t adcCountsInPercents = AdcMeasurements_GetPotentiometerAdcCountsInPercents();
366
367 // Направление изменения громкости звука потенциометром
368 VolumeChangeDirection volumeChangeDirection = INCREASE_VOLUME;
369
370 // Определение изменения положения потенциометра
371 int16_t deltaPotentiometerPosition = (int16_t)adcCountsInPercents - (int16_t)lastPotentiometerPositionInPercents;
372
373 // Если приращение положения потенциометра отрицательное
374 if (deltaPotentiometerPosition < 0)
375 {
376 // Установка направления изменения громкости звука
377 // потенциометром - громкость уменьшается
378 volumeChangeDirection = DECREASE_VOLUME;
379 }
380
381 // Если потенциометр показывает очень низкое значение и
382 // выполняется уменьшение громкости и
383 // громкость регулируется потенциометром
384 if ((adcCountsInPercents < VOLUME_ZERO_THRESHOLD) &&
385 (DECREASE_VOLUME == volumeChangeDirection) &&
386 (VOLUME_CONTROL_SOURCE_POTENTIOMETER == volumeControlSource))
387 {
388 // Обновление последнего положения
389 lastPotentiometerPositionInPercents = adcCountsInPercents;
390
391 // Если громкость не достигла 0%
392 if (0 != currentVolumeInPercents)
393 {
394 // Текущим источником управления
395 // громкостью звука является потенциометр
396 volumeControlSource = VOLUME_CONTROL_SOURCE_POTENTIOMETER;
397
398 // Обновление времени последнего изменения положения потенциометра
399 lastPotentiometerChangePositionTime = UserTimer_GetCounterTime();
400
401 // Установка громкости в 0%
402 currentVolumeInPercents = 0;
403
404 // Синхронизация с переменной смартфона
405 lastPhoneVolumeInPercents = 0;
406
407 // Установка громкости звука
408 volumeStream.SetVolume(currentVolumeInPercents);
409
410 // Расчет громкости звука в рамках протокола AVRCP
411 uint8_t avrcpVolume = (uint8_t) (currentVolumeInPercents * VOLUME_PERCENT_TO_AVRCP_COEFF);
412
413 // Обновление текущей громкости звука на смартфоне
414 a2dp_sink.set_volume(avrcpVolume);
415 }
416
417 // Выход, чтобы не выполнялась двойная обработка
418 return;
419 }
420
421 // Пояснение. Значения АЦП, приходящие с каналов, к которым подключены
422 // потенциометры изменяются в большом диапазоне даже в случае, когда
423 // потенциометр не вращается. Чтобы случайные изменения в значениях АЦП
424 // не приводили к изменению громкости звука, была добавлена защита.
425 // Изменение громкости с потенциометра фиксируется только, когда громкость
426 // изменилась на значение, превышающее DELTA_VOLUME_THRESHOLD_VALUE.
427
428 // Если изменение положения потенциометра превышает пороговое значение
429 if (abs(deltaPotentiometerPosition) > DELTA_VOLUME_THRESHOLD_VALUE)
430 {
431 // Обновление последнего положения потенциометра
432 lastPotentiometerPositionInPercents = adcCountsInPercents;
433
434 // Текущим источником управления
435 // громкостью звука становится потенциометр
436 volumeControlSource = VOLUME_CONTROL_SOURCE_POTENTIOMETER;
437
438 // Сохранение времени последнего
439 // изменения положения потенциометра
440 lastPotentiometerChangePositionTime = UserTimer_GetCounterTime();
441
442 // Обновление текущей громкости звука
443 currentVolumeInPercents = adcCountsInPercents;
444
445 // Ограничение максимальной громкости
446 if (currentVolumeInPercents > MAX_LIMITED_VOLUME_PERCENT)
447 {
448 currentVolumeInPercents = MAX_LIMITED_VOLUME_PERCENT;
449 }
450
451 // Защита от опасной комбинации громкости и пресета
452 VolumeProtectionCheck(currentVolumeInPercents);
453
454 // Установка громкости звука
455 volumeStream.SetVolume(currentVolumeInPercents);
456
457 // Расчет громкости звука в рамках протокола AVRCP
458 uint8_t avrcpVolume = (uint8_t) (currentVolumeInPercents * VOLUME_PERCENT_TO_AVRCP_COEFF);
459
460 // Обновление текущей громкости звука
461 // смартфоне для синхронизации значения
462 a2dp_sink.set_volume(avrcpVolume);
463
464 #ifdef DEBUG_INFO_VOLUME_CONTROL
465
466 Serial.printf("Громкость, установленная потенциометром: %u%%\r\n", currentVolumeInPercents);
467
468 #endif // DEBUG_INFO_VOLUME_CONTROL
469 }
470}

474{

498 SoundControl_SwitchNextTrack();

511 #ifdef DEBUG_INFO_BUTTON_SOUND_CONTROL_STATE

519 pButtonSeriesStatus[BUTTON_SOUND_CONTROL] = BUTTON_PRESS_SERIES_NONE;
520
521 // Сброс количества нажатий на кнопку управления звука
522 pButtonsPressCount[BUTTON_SOUND_CONTROL] = 0;
523 }
