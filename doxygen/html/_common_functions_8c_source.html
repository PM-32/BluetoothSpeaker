18PinState CommonFunctions_GpioGetState(uint8_t gpioPin)
19{
20 // Битовая маска для выделения пина
21 uint32_t pinBitMask = 0;
22
23 // Состояние бита (пина)
24 PinState pinState = PIN_RESET;
25
26 // Если номер пина меньше GPIO_REG_SIZE_IN_BITS
27 if (gpioPin < GPIO_REG_SIZE_IN_BITS)
28 {
29 pinBitMask = (uint32_t) (PIN_SET << gpioPin);
30
31 // Чтение состояния пина из регистра GPIO_IN_REG
32 pinState = REG_READ(GPIO_IN_REG) & pinBitMask;
33 }
34 else // Если номер пина не менее GPIO_REG_SIZE_IN_BITS
35 {
36 // Номер пина не превышает максимальный номер пина
37 if (gpioPin <= MAX_PIN_NUMBER)
38 {
39 pinBitMask = (uint32_t) (PIN_SET << (gpioPin - GPIO_REG_SIZE_IN_BITS));
40
41 // Чтение состояния пина из регистра GPIO_IN1_REG
42 pinState = REG_READ(GPIO_IN1_REG) & pinBitMask;
43 }
44 }
45
46 return pinState;
47}

52void CommonFunctions_GpioSetState(uint8_t gpioPin, PinState pinState)
53{
54 // Битовая маска для выделения пина

56

59 {
60 pinBitMask = (uint32_t) (PIN_SET << gpioPin);
61
62 // Если нужно установить лог. "1"
63 if (PIN_SET == pinState)
64 {
65 // Установка бита в регистре GPIO_OUT_REG
66 REG_WRITE(GPIO_OUT_REG, REG_READ(GPIO_OUT_REG) | pinBitMask);
67 }

69 {
70 // Сброс бита в регистре GPIO_OUT_REG

73 }
