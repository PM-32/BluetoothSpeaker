48void ButtonsDriver_AntibounceFilter(void)
49{
50 // Массив пинов кнопок
51 static uint8_t buttonsPinArray[BUTTONS_QUANTITY] = { BUTTON_SOUND_CONTROL_PIN, \
52 BUTTON_BLUETOOTH_CONNECTION_CONTROL_PIN, \
53 BUTTON_SOUND_PRESET_CONTROL_PIN };
54
55 // Массив с предельными количествами фиксируемых нажатий на кнопки
56 static uint8_t maxButtonsPress[BUTTONS_QUANTITY] = { THREE_PRESS, TWO_PRESS, THREE_PRESS };
57
58 // Текущая стадия работы фильтра антидребезга
59 // кнопки для определения многократных нажатий
60 static volatile SeveralPressFilterStage severalPressFilterStage[BUTTONS_QUANTITY];
61
62 // Статус определения нового нажатия на кнопку
63 static volatile NewPressDetectedStatus newPressDetectedStatus[BUTTONS_QUANTITY];
64
65 // Время, когда был запущен таймер для отсчета
66 // таймаута следующего нажатия на кнопку
67 static volatile uint32_t waitingForNextPressStartTime[BUTTONS_QUANTITY];
68
69 // Счетчик времени, когда кнопка была нажата
70 static volatile uint16_t counterButtonPressed[BUTTONS_QUANTITY];
71
72 // Счетчик времени, когда кнопка была не нажата
73 static volatile uint16_t counterButtonNotPressed[BUTTONS_QUANTITY];
74
75 // Количество зафиксированных устойчивых нажатий на кнопку в момент
76 // запуска таймера для отсчета таймаута следующего нажатия на кнопку
77 static volatile uint8_t waitingStartPress[BUTTONS_QUANTITY];
78
79 for (uint8_t buttonIndex = 0; buttonIndex < BUTTONS_QUANTITY; buttonIndex++)
80 {
81 // Чтение текущего состояния кнопки
82 PinState currentButtonState = CommonFunctions_GpioGetState(buttonsPinArray[buttonIndex]);
83
84 // Если запущен таймер ожидания следующего нажатия на кнопку
85 if (WAITING_FOR_NEXT_PRESS == severalPressFilterStage[buttonIndex])
86 {
87 // Если достигнут таймаут ожидания следующего нажатия на кнопку
88 if (NEXT_BUTTON_PRESS_TIMEOUT <= (UserTimer_GetCounterTime() - waitingForNextPressStartTime[buttonIndex]))
89 {
90 // Если было хотя бы одно нажатие
91 if (buttonsPressCount[buttonIndex] > 0)
92 {
93 // Установка статуса - серия нажатий на кнопку завершена
94 buttonsPressSeriesStatus[buttonIndex] = BUTTON_PRESS_SERIES_FINISHED;
95 }
96
97 // Сброс времени, когда был запущен таймер
98 // для отсчета таймаута следующего нажатия на кнопку
99 waitingForNextPressStartTime[buttonIndex] = 0;
100
101 // Сброс стадии работы фильтра антидребезга
102 // кнопки для определения многократных нажатий
103 severalPressFilterStage[buttonIndex] = START_SEVERAL_PRESS_FILTER;
104 }
105 }
106
107 // Если кнопка нажата
108 if (PIN_RESET == currentButtonState)
109 {
110 // Увеличение счетчика времени,
111 // когда кнопка была нажата
112 counterButtonPressed[buttonIndex]++;
113
114 // Сброс счетчика времени,
115 // когда кнопка была не нажата
116 counterButtonNotPressed[buttonIndex] = 0;
117
118 // Если кнопка была нажата в течение
119 // времени работы фильтра антидребезга
120 if (ANTIBOUNCE_FILTER_IN_TICKS <= counterButtonPressed[buttonIndex])
121 {
122 // Если запущен таймер ожидания следующего нажатия на кнопку
123 if (WAITING_FOR_NEXT_PRESS == severalPressFilterStage[buttonIndex])
124 {
125 // Ограничение максимального количества фиксируемых нажатий на кнопку
126 if (buttonsPressCount[buttonIndex] < maxButtonsPress[buttonIndex])
127 {
128 // Увеличение счетчика нажатий
129 buttonsPressCount[buttonIndex] = waitingStartPress[buttonIndex] + 1;
130 }
131
132 // Установка статуса - новое нажатие на кнопку зафиксировано
133 newPressDetectedStatus[buttonIndex] = NEW_PRESS_DETECTED;
134
135 // Сброс статуса завершения серии нажатий на кнопку
136 buttonsPressSeriesStatus[buttonIndex] = BUTTON_PRESS_SERIES_NONE;
137 }
138 else // Если не запущен таймер ожидания следующего нажатия на кнопку
139 {
140 // Если не было зафиксировано новое нажатие на кнопку
141 if (NEW_PRESS_DETECTED != newPressDetectedStatus[buttonIndex])
142 {
143 // Зафиксировано первое устойчивое нажатие на кнопку
144 buttonsPressCount[buttonIndex] = 1;
145
146 // Сброс статуса завершения серии нажатий на кнопку
147 buttonsPressSeriesStatus[buttonIndex] = BUTTON_PRESS_SERIES_NONE;
148 }
149
150 // Если фильтр антидребезга кнопки для определения
151 // многократного нажатия находится в стадии сброса
152 if (START_SEVERAL_PRESS_FILTER == severalPressFilterStage[buttonIndex])
153 {
154 // Установка стадии - зафиксировано первое устойчивое нажатие на кнопку
155 severalPressFilterStage[buttonIndex] = FIRST_PRESS_DETECTED;
156 }
157 }
158 }
159 }
160 else // Если кнопка не нажата
161 {
162 // Увеличение счетчика времени,
163 // когда кнопка была не нажата
164 counterButtonNotPressed[buttonIndex]++;
165
166 // Сброс счетчика времени,
167 // когда кнопка была нажата
168 counterButtonPressed[buttonIndex] = 0;
169
170 // Если кнопка была не нажата в течение
171 // времени работы фильтра антидребезга
172 if (ANTIBOUNCE_FILTER_IN_TICKS <= counterButtonNotPressed[buttonIndex])
173 {
174 // Если зафиксировано первое устойчивое нажатие на
175 // кнопку или зафиксировано новое нажатие на кнопку
176 if ((FIRST_PRESS_DETECTED == severalPressFilterStage[buttonIndex]) ||
177 (NEW_PRESS_DETECTED == newPressDetectedStatus[buttonIndex]))
178 {
179 // Установка стадии - ожидание следующего нажатия на кнопку
180 // (запуск таймера для отсчета таймаута следующего нажатия на кнопку)
181 severalPressFilterStage[buttonIndex] = WAITING_FOR_NEXT_PRESS;
182
183 // Сохранение времени, когда был запущен таймер
184 // для отсчета таймаута следующего нажатия на кнопку
185 waitingForNextPressStartTime[buttonIndex] = UserTimer_GetCounterTime();
186
187 // Сохранение количества нажатий на кнопку в момент запуска
188 // таймера для отсчета таймаута следующего нажатия на кнопку
189 waitingStartPress[buttonIndex] = buttonsPressCount[buttonIndex];
190
191 // Сброс статуса определения нового нажатия на кнопку
192 newPressDetectedStatus[buttonIndex] = NEW_PRESS_NOT_DETECTED;
193 }
194 }
195 }
196 }
197}
