58void AmplifierAndDacControl_TurnOnChipsAfterDelay(void)
59{
60 // Статус включения усилителя
61 static AmplifierStatus amplifierStatus = AMPLIFIER_OFF;
62
63 // Статус включения звука через ЦАП
64 static SoundStatus soundStatus = SOUND_MUTE;
65
66 // Если звук на ЦАП ещё не включен
67 if (SOUND_MUTE == soundStatus)
68 {
69 // Выжидание задержки
70 if (UserTimer_GetCounterTime() > DELAY_BEFORE_SET_SOUND_ON)
71 {
72 // Включение звука через ЦАП
73 SOUND_UNMUTE();
74
75 // Установка статуса - звук уже включен
76 soundStatus = SOUND_ALREADY_UNMUTE;
77 }
78 }
79
80 // Если усилитель ещё не включен
81 // и звук включен через ЦАП включен
82 if ((AMPLIFIER_OFF == amplifierStatus) &&
83 (SOUND_ALREADY_UNMUTE == soundStatus))
84 {
85 // Выжидание задержки
86 if (UserTimer_GetCounterTime() > (DELAY_BEFORE_SET_SOUND_ON + DELAY_BEFORE_AMPLIFIER_POWER_ON))
87 {
88 // Включение усилителя
89 AMPLIFIER_TURN_ON();
90
91 // Установка статуса - усилитель уже включен
92 amplifierStatus = AMPLIFIER_ALREADY_ON;
93 }
94 }
