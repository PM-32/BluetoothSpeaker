58static float CalculateImpulseEnvelope(float time, float startTime, float endTime)
59{
60 // Если текущее время не попадает в интервал импульса
61 if ((time < startTime) || (time >= endTime))
62 {
63 return 0.0f;
64 }
65
66 // Нормированное время внутри импульса (0.0 - 1.0)
67 float normalizedTime = (time - startTime) / (endTime - startTime);
68
69 // Примечание. Огибающая имеет колоколообразную форму: sin(pi * time)^2.
70 // Возведение в квадрат обеспечивает плавное нарастание и затухание,
71 // а также отсутствие резких скачков на границах импульса
72
73 // Расчет огибающей
74 float envelope = sinf(pi * normalizedTime);
75
76 // Возведение огибающей в квадрат
77 float squareEnvelope = envelope * envelope;
78
79 return squareEnvelope;
80}

99static void GenerateTone(int16_t *samplesBuffer, uint32_t samplesQuantity, uint32_t startSample)
100{
101 // Выбор частоты сигнала в зависимости от типа уведомления
102 float frequency = (player.currentType == NOTIFICATION_CONNECT) ? CONNECT_FREQUENCY : DISCONNECT_FREQUENCY;
103
104 for (uint32_t sampleIndex = 0; sampleIndex < samplesQuantity; sampleIndex++)
105 {
106 // Текущее абсолютное время
107 float currentTime = (float) (startSample + sampleIndex) / SAMPLE_RATE;
108
109 // Если время вышло за пределы длительности,
110 // то ничего не воспроизводится
111 if (currentTime >= NOTIFICATION_DURATION_SEC)
112 {
113 samplesBuffer[sampleIndex] = 0;
114
115 continue;
116 }
117
118 // Расчет огибающей из двух импульсов
119 float envelope = 0.0f;
120
121 // Расчет огибающей первого импульса
122 envelope += CalculateImpulseEnvelope(currentTime, FIRST_IMPULSE_START, FIRST_IMPULSE_END);
123
124 // Расчет огибающей второго импульса
125 envelope += CalculateImpulseEnvelope(currentTime, SECOND_IMPULSE_START, SECOND_IMPULSE_END);
126
127 // Ограничение суммарной огибающей
128 if (envelope > IMPULSE_MAX_AMPLITUDE)
129 {
130 envelope = IMPULSE_MAX_AMPLITUDE;
131 }
132
133 // Пояснение. Расчет приращения фазы для текущей частоты.
134 // deltaPhase = 2 * pi * frequency / NOTIFICATION_SAMPLE_RATE,
135 // где deltaPhase - приращение частоты фазы;
136 // frequency - частота сигнала;
137 // NOTIFICATION_SAMPLE_RATE - частота дискретизации.
138 // При этом (2 * pi) радиан составляют один полный период синусоиды.
139
140 // Расчет приращения фазы для текущей частоты
141 float deltaPhase = PHASE_CYCLE * pi * frequency / SAMPLE_RATE;
142
143 // Текущая фаза синусоиды
144 player.phase += deltaPhase;
145
146 // Нормализация фазы
147 if (player.phase > PHASE_CYCLE * pi)
148 {
149 player.phase -= PHASE_CYCLE * pi;
150 }
151
152 // Генерация синусоидального сигнала с огибающей
153 float sampleValue = NOTIFICATION_AMPLITUDE * envelope * sinf(player.phase);
154
155 // Ограничение громкости для 16-ти битного сэмпла
156 if (sampleValue > MAX_VOLUME_SAMPLE)
157 {
158 sampleValue = MAX_VOLUME_SAMPLE;
159 }
160 else if (sampleValue < MIN_VOLUME_SAMPLE)
161 {
162 sampleValue = MIN_VOLUME_SAMPLE;
163 }
164
165 // Сохранение полученного значения в буфер
166 samplesBuffer[sampleIndex] = (int16_t) sampleValue;
167 }

187void AudioNotifications_Play(NotificationType notificationType)

198 player.phase = 0; // Текущая фаза синусоиды

211void AudioNotifications_Update(void)
212{
213 // Выход, если воспроизведение не активно
214 if (NOTIFICATION_PLAYING != player.playbackState)
215 {
216 return;
217 }
218
219 // Определение количества сэмплов, которое осталось отправить
220 uint32_t remainingSamples = NOTIFICATION_TOTAL_SAMPLES - player.samplesSent;
221
222 // Определение размера буфера данных для генерации
223 uint32_t samplesBufferSize = NOTIFICATION_BUFFER_SIZE;
224
225 // Если осталось передать меньше данных,
226 // чем размер буфера, то происходит уменьшение буфера
227 if (samplesBufferSize > remainingSamples)
228 {
229 samplesBufferSize = remainingSamples;
230 }
231
232 // Буфер для генерации сэмплов (моно)
233 int16_t monoBuffer[NOTIFICATION_BUFFER_SIZE];
234
235 // Генерация звука (сэмплов)
236 GenerateTone(monoBuffer, samplesBufferSize, player.samplesSent);
237
238 // Буфер для стерео звука
239 int16_t stereoBuffer[NOTIFICATION_BUFFER_SIZE * AUDIO_CHANNELS_QUANTITY];
240
241 // Преобразование звука из моно в стерео
242 MonoToStereoConvert(monoBuffer, stereoBuffer, samplesBufferSize);
243
244 // Получение адреса объекта I2SStream
245 I2SStream *i2s = SoundControl_GetI2SStreamPointer();
246
247 // Если указатель существует
248 if (NULL != i2s)
249 {
250 // Вывод звука через I2S
251 i2s->write((uint8_t *) stereoBuffer, samplesBufferSize * BYTES_IN_STEREO_SAMPLE);
252 }
253
254 // Инкремент количества отправленных сэмплов
255 player.samplesSent += samplesBufferSize;
256
257 // Если все сэмплы отправлены
258 if (player.samplesSent >= NOTIFICATION_TOTAL_SAMPLES)
259 {
260 // Установка состояния - уведомление не воспроизводится
261 player.playbackState = NOTIFICATION_NOT_PLAYING;
262
263 #ifdef DEBUG_INFO_AUDIO_NOTIFICATIONS
264
265 Serial.println("Воспроизведение завершено");
266
267 #endif // DEBUG_INFO_AUDIO_NOTIFICATIONS
268 }
