84static float CalculateBatteryVoltageInPercents(float batVoltage)
85{
86 // Проверка пределов напряжения батареи (0 - 100 %)
87 if (batVoltage > BATTERY_MAX_VOLTAGE)
88 {
89 return VOLTAGE_IN_PERCENTS_MAX;
90 }
91 else if (batVoltage < BATTERY_MIN_VOLTAGE)
92 {
93 return VOLTAGE_IN_PERCENTS_MIN;
94 }
95
96 // Заряд батареи определяется по кусочно-линейной функции.
97 // Это обосновывается тем, что напряжение начинает падать быстрее при
98 // большом разряде батареи. Участок 100 - 15% рассчитывается по одной
99 // формуле, участок 15 - 0% по другой. Таким образом, точка 15% делит
100 // расчет на две части.
101
102 // Напряжение батареи в процентах
103 float voltageInPercents = 0.0f;
104
105 if (batVoltage >= BATTERY_LAST_15_PROCENTS_VOLTAGE) // Расчет заряда для участка 100 - 15%
106 {
107 voltageInPercents = BATTERY_LAST_15_PROCENTS + (batVoltage - BATTERY_LAST_15_PROCENTS_VOLTAGE) * \
108 (BATTERY_85_PROCENTS / (BATTERY_MAX_VOLTAGE - BATTERY_LAST_15_PROCENTS_VOLTAGE));
109 }
110 else // Расчет заряда для участка 15 - 0%
111 {
112 voltageInPercents = (batVoltage - BATTERY_MIN_VOLTAGE) * (BATTERY_LAST_15_PROCENTS / \
113 (BATTERY_LAST_15_PROCENTS_VOLTAGE - BATTERY_MIN_VOLTAGE));
114 }
115
116 #ifdef DEBUG_INFO_BATTERY_VOLTAGE
117
118 // Время последней отправки сообщения в терминал
119 static uint32_t lastMessageSendTime = 0;
120
121 // Вывод сообщения с периодом MONITORING_MESSAGE_PERIOD
122 if ((UserTimer_GetCounterTime() - lastMessageSendTime) > MONITORING_MESSAGE_PERIOD)
123 {
124 Serial.printf("Заряд батареи: %.1f %%\r\n", voltageInPercents);
125 Serial.print("---------------------------\r\n");
126
127 // Обновление времени последней отправки сообщения в терминал
128 lastMessageSendTime = UserTimer_GetCounterTime();
129 }
130
131 #endif // DEBUG_INFO_BATTERY_VOLTAGE
132
133 // Ограничение пределов заряда батареи в процентах (0 - 100 %)
134 if (voltageInPercents > VOLTAGE_IN_PERCENTS_MAX)
135 {
136 return VOLTAGE_IN_PERCENTS_MAX;
137 }
138 else if (voltageInPercents < VOLTAGE_IN_PERCENTS_MIN)
139 {
140 return VOLTAGE_IN_PERCENTS_MIN;
141 }
142
143 return voltageInPercents;
144}

149static uint8_t GetColorIndexWithHysteresis(float batteryVoltageInPercents)
150{
151 // Последний установленный индекс цвета
152 static uint8_t lastColorIndex = 0;
153
154 // Последний измеренный процент заряда батареи
155 static float lastBatteyVoltageInPercents = -1.0f;
156
157 // Первый вызов функции - вычисление индекса и
158 // сохранение процента заряда батареи
159 if (lastBatteyVoltageInPercents < 0.0f)
160 {
161 // Сохранение последнего процента заряда батареи
162 lastBatteyVoltageInPercents = batteryVoltageInPercents;
163
164 // Сохранение индекса последнего цвета
165 lastColorIndex = (uint8_t) (batteryVoltageInPercents / COLOR_CHANGE_INTERVAL_IN_PERCENTS);
166
167 // Ограничение индекса последнего цвета
168 if (lastColorIndex >= COLORS_QUANTITY)
169 {
170 lastColorIndex = COLORS_QUANTITY - 1;
171 }
172
173 return lastColorIndex;
174 }
175
176 // Определение "сырого" индекса цвета
177 uint8_t basicColorIndex = (uint8_t) (batteryVoltageInPercents / COLOR_CHANGE_INTERVAL_IN_PERCENTS);
178
179 // Ограничение "сырого" индекса цвета
180 if (basicColorIndex >= COLORS_QUANTITY)
181 {
182 basicColorIndex = COLORS_QUANTITY - 1;
183 }
184
185 // Если "сырой" индекс совпадает
186 // с текущим, то ничего не меняется
187 if (basicColorIndex == lastColorIndex)
188 {
189 // Сохранение последнего процента заряда батареи
190 lastBatteyVoltageInPercents = batteryVoltageInPercents;
191
192 return lastColorIndex;
193 }
194
195 if (basicColorIndex < lastColorIndex) // Если заряда становится меньше - переключение по порогу на следующий цвет
196 {
197 // Расчет порога для переключения цвета с учетом гистерезиса
198 float thresholdColorChangeInPercents = (float) ((lastColorIndex * COLOR_CHANGE_INTERVAL_IN_PERCENTS) - COLOR_CHANGE_HYSTERESIS_IN_PERCENTS);
199
200 // Проверка достижения порога для переключения цвета вниз
201 if (batteryVoltageInPercents <= thresholdColorChangeInPercents)
202 {
203 lastColorIndex = basicColorIndex;
204 }
205 }
206 else if (basicColorIndex > lastColorIndex) // Если заряда становится больше - переключение по порогу на следующий цвет
207 {
208 // Расчет порога для переключения цвета с учетом гистерезиса
209 float thresholdColorChangeInPercents = (float) ((basicColorIndex * COLOR_CHANGE_INTERVAL_IN_PERCENTS) + COLOR_CHANGE_HYSTERESIS_IN_PERCENTS);
210
211 // Проверка достижения порога для переключения цвета вниз
212 if (batteryVoltageInPercents >= thresholdColorChangeInPercents)
213 {
214 lastColorIndex = basicColorIndex;
215 }
216 }
217
218 // Сохранение последнего процента заряда батареи
219 lastBatteyVoltageInPercents = batteryVoltageInPercents;
220
221 return lastColorIndex;
222}

225void BatteryChargeIndication_Execute(void)
226{
227 // Получение текущего напряжения батареи
228 float currentBatteryVoltage = AdcMeasurements_GetBatteryVoltage();
229
230 #ifdef DEBUG_INFO_BATTERY_VOLTAGE
231
232 // Время последней отправки сообщения в терминал
233 static uint32_t lastMessageSendTime = 0;
234
235 // Вывод сообщения с периодом MONITORING_MESSAGE_PERIOD
236 if ((UserTimer_GetCounterTime() - lastMessageSendTime) > MONITORING_MESSAGE_PERIOD)
237 {
238 Serial.printf("Напряжение батареи: %.3f В\r\n", currentBatteryVoltage);
239
240 // Обновление времени последней отправки сообщения в терминал
241 lastMessageSendTime = UserTimer_GetCounterTime();
242 }
243
244 #endif // DEBUG_INFO_BATTERY_VOLTAGE
245
246 // Определение заряда батареи в процентах
247 float currentBatteryVoltageInPercents = CalculateBatteryVoltageInPercents(currentBatteryVoltage);
248
249 // Определение индекса цвета для индикации заряда с учетом гистерезиса
250 uint8_t colorIndex = GetColorIndexWithHysteresis(currentBatteryVoltageInPercents);
251
252 if (colorIndex < COLORS_QUANTITY) // Проверка на допустимость индекса цвета
253 {
254 // Включение светодиода необходимым цветом
255 BatteryChargeIndication_SetColor(colors[colorIndex]);
256 }
257 else // Выключение светодиода, если индекс цвета не корректен
258 {
259 BatteryChargeIndication_SetColor(ledOff);
260 }
261}
