102static float CalibrateBatteryVoltage(float rawVoltage)
103{
104 // Если напряжение выше первой точки таблицы - экстраполяция вверх
105 if (rawVoltage >= calibrationTable[0].adcVoltage)
106 {
107 // Вычисление коэффициента наклона по двум первым точкам
108 float slope = (calibrationTable[1].realVoltage - calibrationTable[0].realVoltage) /
109 (calibrationTable[1].adcVoltage - calibrationTable[0].adcVoltage);
110
111 // Экстраполяция напряжения
112 float calibratedVoltage = calibrationTable[0].realVoltage +
113 slope * (rawVoltage - calibrationTable[0].adcVoltage);
114
115 return calibratedVoltage;
116 }
117
118 // Если напряжение ниже последней точки таблицы - экстраполяция вниз
119 if (rawVoltage <= calibrationTable[CALIBRATION_POINTS_QUANTITY - 1].adcVoltage)
120 {
121 // Вычисление коэффициента наклона по двум последним точкам
122 float slope = (calibrationTable[CALIBRATION_POINTS_QUANTITY - 1].realVoltage -
123 calibrationTable[CALIBRATION_POINTS_QUANTITY - 2].realVoltage) /
124 (calibrationTable[CALIBRATION_POINTS_QUANTITY - 1].adcVoltage -
125 calibrationTable[CALIBRATION_POINTS_QUANTITY - 2].adcVoltage);
126
127 // Экстраполяция напряжения
128 float calibratedVoltage = calibrationTable[CALIBRATION_POINTS_QUANTITY - 1].realVoltage +
129 slope * (rawVoltage - calibrationTable[CALIBRATION_POINTS_QUANTITY - 1].adcVoltage);
130
131 return calibratedVoltage;
132 }
133
134 // Поиск интервала для интерполяции
135 for (uint8_t pointIndex = 0; pointIndex < (CALIBRATION_POINTS_QUANTITY - 1); pointIndex++)
136 {
137 // Если текущее напряжение попадает в интервал между calibrationTable[pointIndex + 1] и calibrationTable[pointIndex]
138 if ((rawVoltage >= calibrationTable[pointIndex + 1].adcVoltage) &&
139 (rawVoltage < calibrationTable[pointIndex].adcVoltage))
140 {
141 // Вычисление коэффициента интерполяции
142 float interpolationCoefficient = (rawVoltage - calibrationTable[pointIndex + 1].adcVoltage) /
143 (calibrationTable[pointIndex].adcVoltage - calibrationTable[pointIndex + 1].adcVoltage);
144
145 // Вычисление разницы напряжений
146 float voltageDifference = calibrationTable[pointIndex].realVoltage - calibrationTable[pointIndex + 1].realVoltage;
147
148 // Линейная интерполяция напряжения
149 float calibratedVoltage = calibrationTable[pointIndex + 1].realVoltage +
150 interpolationCoefficient * voltageDifference;
151
152 return calibratedVoltage;
153 }
154 }
155
156 return rawVoltage;
157}

160static void AdcMeasurements_MovingAverageFilter(void)
161{
162 // Буферы для хранения измерений АЦП для каждого канала отдельно
163 static uint16_t adcValuesPotentiometerChannel[ADC_MEASUREMENTS_QUANTITY_POTENTIOMETER];
164 static uint16_t adcValuesBatteryChannel[ADC_MEASUREMENTS_QUANTITY_BATTERY];
165
166 // Суммарное значение АЦП канала
167 static uint32_t sumAdcValue[ADC_CHANNELS_QUANTITY];
168
169 // Индекс текущего измерения АЦП
170 static uint8_t measurementIndex[ADC_CHANNELS_QUANTITY];
171
172 // Счетчик заполненных измерений в буфере АЦП
173 // (счетчик заходов в функцию)
174 static uint8_t measurementsCount[ADC_CHANNELS_QUANTITY];
175
176 for (uint8_t indexChannel = 0; indexChannel < ADC_CHANNELS_QUANTITY; indexChannel++)
177 {
178 // Размер буфера для хранения измерений АЦП канала
179 uint8_t channelMeasurementsQuantuty = 1;
180
181 // Указатель на буфер для хранения измерений АЦП канала
182 uint16_t *pAdcValuesChannel = NULL;
183
184 // В зависимости от канала выбирается указатель
185 // на необходимый буфер и количество измерений АЦП
186 if (ADC_POTENTIOMETER_CHANNEL_INDEX == indexChannel)
187 {
188 channelMeasurementsQuantuty = ADC_MEASUREMENTS_QUANTITY_POTENTIOMETER;
189 pAdcValuesChannel = (uint16_t *) &adcValuesPotentiometerChannel;
190 }
191 else if (ADC_BATTERY_CHANNEL_INDEX == indexChannel)
192 {
193 channelMeasurementsQuantuty = ADC_MEASUREMENTS_QUANTITY_BATTERY;
194 pAdcValuesChannel = (uint16_t *) &adcValuesBatteryChannel;
195 }
196
197 // Получение текущего значения канала АЦП
198 uint16_t currentAdcValue = (uint16_t) analogRead(adcPins[indexChannel]);
199
200 // Вычитание старого значения из суммарного
201 // значения АЦП, если буфер уже заполнен
202 if (measurementsCount[indexChannel] >= channelMeasurementsQuantuty)
203 {
204 sumAdcValue[indexChannel] -= pAdcValuesChannel[measurementIndex[indexChannel]];
205 }
206
207 // Сохранение нового значения АЦП
208 pAdcValuesChannel[measurementIndex[indexChannel]] = currentAdcValue;
209
210 // Добавление нового значения в суммарное значение АЦП
211 sumAdcValue[indexChannel] += currentAdcValue;
212
213 // Увеличение индекса измерений АЦП
214 measurementIndex[indexChannel]++;
215
216 // Сброс индекса измерений АЦП
217 if (measurementIndex[indexChannel] >= channelMeasurementsQuantuty)
218 {
219 measurementIndex[indexChannel] = 0;
220 }
221
222 // Увеличение счетчика заполненных измерений
223 if (measurementsCount[indexChannel] < channelMeasurementsQuantuty)
224 {
225 measurementsCount[indexChannel]++;
226 }
227
228 // Если выполнено, как минимум, одно измерение АЦП
229 if (measurementsCount[indexChannel] > 0)
230 {
231 // Вычисление усредненного значения отсчетов АЦП
232 uint16_t meanAdcValue = (uint16_t) (sumAdcValue[indexChannel] / measurementsCount[indexChannel]);
233
234 if (ADC_POTENTIOMETER_CHANNEL_INDEX == indexChannel) // Канал АЦП для потенциометра
235 {
236 // Вычисление усредненного значения отсчетов АЦП с калибровкой
237 uint16_t meanCalibratedAdcCounts = (uint16_t) (POTENTIOMETER_VOLUME_CONTROL_C0 * meanAdcValue + POTENTIOMETER_VOLUME_CONTROL_C1);
238
239 // Ограничение отсчетов АЦП
240 if (meanCalibratedAdcCounts > ADC_MAX_COUNTS)
241 {
242 meanCalibratedAdcCounts = ADC_MAX_COUNTS;
243 }
244
245 // Вычисление отсчетов АЦП в процентах
246 potentiometerAdcCountsInPercents = (uint8_t) (meanCalibratedAdcCounts * ADC_COUNTS_TO_PERCENTS_COEFF);
247
248 // Ограничение процентов
249 if (potentiometerAdcCountsInPercents > MAX_PERCENTS)
250 {
251 potentiometerAdcCountsInPercents = MAX_PERCENTS;
252 }
253 }
254 else if (ADC_BATTERY_CHANNEL_INDEX == indexChannel) // Канал АЦП для батареи
255 {
256 // Фактическое измеренное напряжение батареи
257 float rawBatteryVoltage = (float)(meanAdcValue * ADC_COUNTS_TO_VOLTAGE_COEFF * INVERSE_RESISTIVE_DIVIDER_COEFF);
258
259 // Откалиброванное напряжение батареи
260 batteryVoltage = CalibrateBatteryVoltage(rawBatteryVoltage);
261 }
262 }
263 }
264}

267void AdcMeasurements_Pollings(void)
268{
269 // Время последнего опроса АЦП
270 static uint32_t lastAdcPollingTime = 0;
271
272 // Чтение текущего положения ручки потенциометра
273 // каждые ADC_POLLINGS_PERIOD периодов таймера 0
274 if ((UserTimer_GetCounterTime() - lastAdcPollingTime) > ADC_POLLINGS_PERIOD)
275 {
276 // Фильтр скользящего среднего для канала АЦП
277 AdcMeasurements_MovingAverageFilter();
278
279 // Обновление времени последнего опроса АЦП
280 lastAdcPollingTime = UserTimer_GetCounterTime();
281 }
