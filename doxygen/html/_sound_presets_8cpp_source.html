139static float ConvertDbToLinearCoefficient(int8_t dB)
140{
141 // Пояснение. Формула преобразования: linear = 10^(dB/20),
142 // где 10 - основание десятичного логарифма
143 // 20 - коэффициент для амплитуды (децибелы для
144 // амплитуды считаются как 20 * log10(A/A0))
145 // Примеры: 0 dB → 1.0 (единичное усиление)
146 // 6 dB → 2.0 (усиление в 2 раза)
147 // -6 dB → 0.5 (ослабление в 2 раза)
148
149 return powf(DECIBEL_TO_LINEAR_BASE, dB / DECIBEL_DIVISOR);
150}

174static float LowPassFilterFirstOrder(float inputValue, float alphaCoeff, float *previousValue)
175{
176 // Пояснение. Формула ФНЧ первого порядка:
177 // y[n] = alpha * x[n] + (1 - alpha) * y[n-1],
178 // где y[n] - текущее выходное значение,
179 // y[n-1] - предыдущее выходное значение,
180 // x[n] - текущее входное значение,
181 // alpha - коэффициент фильтра (чем меньше, тем сильнее сглаживание)
182
183 float outputValue = alphaCoeff * inputValue + (UNITY_GAIN - alphaCoeff) * (*previousValue);
184
185 // Обновление предыдущего значения фильтра для следующего вызова
186 *previousValue = outputValue;
187
188 return outputValue;
189}

196static float HighPassFilterFirstOrder(float inputValue, float alphaCoeff, float *previousValue)
197{
198 // Пояснение. Формула ФВЧ первого порядка:
199 // y[n] = alpha * (x[n] - x[n-1] + y[n-1])
200 // где y[n] - текущее выходное значение,
201 // y[n-1] - предыдущее выходное значение,
202 // x[n] - текущее входное значение,
203 // x[n-1] - предыдущее входное значение,
204 // alpha - коэффициент фильтра (чем меньше, тем сильнее сглаживание)
205
206 float outputValue = alphaCoeff * (inputValue - (*previousValue) + outputValue);
207
208 // Обновление предыдущего значения фильтра для следующего вызова
209 *previousValue = inputValue;
210
211 return outputValue;
212}

218static int16_t ProcessChannel(int16_t sample, FilterState *state)
219{
220 // Текущие значения частот
221 float currentEqualizerBands[EQUALIZER_BANDS_QUANTITY] = { 0.0f, 0.0f, 0.0f };
222
223 // Обработанные значения частот
224 float processedEqualizerBands[EQUALIZER_BANDS_QUANTITY] = { 0.0f, 0.0f, 0.0f };
225
226 // Обработанный сэмпл
227 float outputSample = 0.0f;
228
229 // Преобразование входного сэмпла из целочисленного в вещественный тип
230 float inputSample = (float) sample;
231
232 // Выделение низких частот с помощью ФНЧ
233 currentEqualizerBands[LOW_FREQUENCIES] = LowPassFilterFirstOrder(inputSample, lowAlphaCoeff, &state->lowFrequencyPrevious);
234
235 // Выделение высоких частот с помощью ФВЧ
236 currentEqualizerBands[HIGH_FREQUENCIES] = HighPassFilterFirstOrder(inputSample, highAlphaCoeff, &state->highFrequencyPrevious);
237
238 // Выделение средних частот (исходный сигнал минус низкие и высокие частоты)
239 currentEqualizerBands[MID_FREQUENCIES] = inputSample - currentEqualizerBands[LOW_FREQUENCIES] - currentEqualizerBands[HIGH_FREQUENCIES];
240
241 for (uint8_t equalizerBandIndex = 0; equalizerBandIndex < EQUALIZER_BANDS_QUANTITY; equalizerBandIndex++)
242 {
243 // Применение усиления к каждой полосе
244 processedEqualizerBands[equalizerBandIndex] = currentEqualizerBands[equalizerBandIndex] * frequenciesGainCoeff[equalizerBandIndex];
245
246 // Смешивание обработанных полос (суммирование всех частотных компонент)
247 outputSample += processedEqualizerBands[equalizerBandIndex];
248 }
249
250 // Ограничение громкости для 16-ти битного сэмпла
251 if (outputSample > MAX_VOLUME_SAMPLE)
252 {
253 outputSample = MAX_VOLUME_SAMPLE;
254 }
255 else if (outputSample < MIN_VOLUME_SAMPLE)
256 {
257 outputSample = MIN_VOLUME_SAMPLE;
258 }
259
260 // Приведение типа сэмпла обратно к целочисленному
261 return (int16_t) outputSample;
262}

265void SoundPresets_Init(void)
266{
267 // Пояснение. Расчет аргумента экспоненты для фильтра низких/высоких частот:
268 // -2 * pi * Fc / Fs. Отрицательный знак нужен для получения exp(-x)
269 // при вычислении alpha = 1 - exp(-x)
270
271 float omegaLow = -PHASE_CYCLE * pi * LOW_CROSSOVER_FREQUENCY / SAMPLE_RATE;
272 float omegaHigh = -PHASE_CYCLE * pi * HIGH_CROSSOVER_FREQUENCY / SAMPLE_RATE;
273
274 // Пояснение. Расчет коэффициента альфа для ФНЧ по формуле:
275 // alpha = 1 - exp(omegaLow), где 1 - единица, обеспечивающая значение
276 // alpha в диапазоне (0, 1), exp(omegaLow) - экспонента отрицательного
277 // аргумента (значение от 0 до 1). Для ФВЧ аналогично.
278
279 lowAlphaCoeff = ALPHA_COEFF_ONE - expf(omegaLow);
280 highAlphaCoeff = ALPHA_COEFF_ONE - expf(omegaHigh);
281
282 #ifdef DEBUG_INFO_SOUND_PRESETS
283
284 Serial.printf("Инициализация: коэффициент ФНЧ = %.4f, коэффициент ФВЧ = %.4f\r\n", lowAlphaCoeff, highAlphaCoeff);
285
286 #endif // DEBUG_INFO_SOUND_PRESETS
287
288 // Чтение сохраненного пресета из flash-памяти
289 currentPreset = ReadSoundPresetFromFlash();
290
291 // Обновление коэффициентов усиления в соответствии с загруженным пресетом
292 SoundPresets_UpdateGains();
293}
