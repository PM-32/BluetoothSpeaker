83void BluetoothConnectionControl_IndicateConnectionStatus(void)
84{
85 // Время последней смены состояния светодиода
86 static uint32_t lastLedChangeStateTime = 0;
87
88 // Текущее состояние Bluetooth-подключения
89 static BluetoothConnectionState connectionState = BLUETOOTH_DISCONNECTED;
90
91 // Получение адреса объекта BluetoothA2DPSink
92 BluetoothA2DPSink *pBluetoothA2DPSink = SoundControl_GetA2DPSinkPointer();
93
94 // Получение нового состояния подключения
95 BluetoothConnectionState newConnectionState = (BluetoothConnectionState) pBluetoothA2DPSink->is_connected();
96
97 // Если состояние подключения изменилось
98 if (newConnectionState != connectionState)
99 {
100 // Обновление текущего состояния подключения
101 connectionState = newConnectionState;
102
103 // Если подключение установлено
104 if (BLUETOOTH_CONNECTED == connectionState)
105 {
106 // Включение светодиода
107 LedsDriver_SetLedState(BLUETOOTH_STATUS_LED_PIN, LED_ON);
108
109 // Воспроизведение звукового уведомления - подключение
110 AudioNotifications_Play(NOTIFICATION_CONNECT);
111 }
112 else // Если подключение не установлено
113 {
114 // Воспроизведение звукового уведомления - отключение
115 AudioNotifications_Play(NOTIFICATION_DISCONNECT);
116 }
117
118 // Обновление времени последней смены состояния светодиода
119 lastLedChangeStateTime = UserTimer_GetCounterTime();
120
121 #ifdef DEBUG_INFO_BLUETOOTH_CONNECTION
122
123 Serial.printf("Состояние соединения изменилось: %s\r\n",
124 (BLUETOOTH_CONNECTED == connectionState) ? "подключено" : "отключено");
125
126 #endif // DEBUG_INFO_BLUETOOTH_CONNECTION
127 }
128
129 // Если подключение отсутствует
130 if (BLUETOOTH_DISCONNECTED == connectionState)
131 {
132 // Переключение состояния светодиода при
133 // достижении периода LED_BLINK_PERIOD_DISCONNECTED
134 if ((UserTimer_GetCounterTime() - lastLedChangeStateTime) > LED_BLINK_PERIOD_DISCONNECTED)
135 {
136 // Смена состояния светодиода
137 LedsDriver_ToggleLed(BLUETOOTH_STATUS_LED_PIN);
138
139 // Обновление времени последней смены состояния светодиода
140 lastLedChangeStateTime = UserTimer_GetCounterTime();
141 }
142 }
143}
